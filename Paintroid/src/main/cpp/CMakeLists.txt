cmake_minimum_required(VERSION 3.22)
project(anpaint_codec LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature TOOLS DEVTOOLS DOXYGEN MANPAGES BENCHMARK EXAMPLES JNI SJPEG OPENEXR VIEWERS TCMALLOC PLUGINS TRANSCODE_JPEG)
  set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../../build/jxl-source jxl EXCLUDE_FROM_ALL)
add_library(anpaint_jxl SHARED jxl_bridge.cpp)
target_link_libraries(anpaint_jxl PRIVATE jxl jnigraphics log)
target_link_options(anpaint_jxl PRIVATE "-Wl,-z,max-page-size=16384")
